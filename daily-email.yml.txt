name: Daily Email

on:
  schedule:
    # Runs every day at 08:00 UTC. Change the time as you like (this is UTC, not your local time).
    - cron: "0 8 * * *"
  workflow_dispatch: # lets you trigger it manually from the Actions tab to test it

permissions:
  contents: write # needed so the workflow can save state.json back to the repo

jobs:
  send-email:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Send the email / check for reply
        env:
          SENDER_EMAIL: ${{ secrets.SENDER_EMAIL }}
          SENDER_PASSWORD: ${{ secrets.SENDER_PASSWORD }}
          RECEIVER_EMAIL: ${{ secrets.RECEIVER_EMAIL }}
          MESSAGE_TEXT: ${{ secrets.MESSAGE_TEXT }}
          START_DATE: ${{ secrets.START_DATE }}
          NTFY_TOPIC: ${{ secrets.NTFY_TOPIC }}
        run: python send_email.py

      - name: Save state if it changed
        run: |
          if [ -n "$(git status --porcelain state.json)" ]; then
            git config user.name "github-actions[bot]"
            git config user.email "github-actions[bot]@users.noreply.github.com"
            git add state.json
            git commit -m "Update state"
            git push
          fi
